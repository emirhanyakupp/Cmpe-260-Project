(* basics.txt - Variable declarations, arithmetic, comparisons, print *)
let a = 15;
let b = 4;
let c = a / b; (* Evaluates to 3, integer division truncated toward zero 
let isValid = (a > 10) and (c == 3);

print(c);        (* Expected output: 3 *)
print(isValid);  (* Expected output: true *)







(* closures.txt - Function returning a function capturing outer variable  *)
let make_multiplier = fun (factor) ->
    fun (x) -> 
        x * factor 
    end
end;

let times_three = make_multiplier(3);
print(times_three(10)); (* Expected output: 30 *)





(* higher_order.txt - Takes another function as argument *)
let apply_twice = fun (f, val) ->
    f(f(val))
end;

let add_five = fun (x) -> x + 5 end;
print(apply_twice(add_five, 10)); (* Expected output: 20 *)







(* recursion.txt - Recursive function demonstrating base case  *)
let factorial = fun (n) ->
    if n <= 1 then 
        1
    else 
        n * factorial(n - 1)
    end
end;

print(factorial(5)); (* Expected output: 120 *)






(* scope.txt - Demonstrates static scoping preserving defining environment  *)
let x = 100;
let get_x = fun () -> x end;
let test_scope = fun () ->
    let x = 200; 
    get_x() (* Resolves to 100 due to static scoping  *)
end;

print(test_scope()); (* Expected output: 100 *)
